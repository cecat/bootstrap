# The MCP server itself, from PyPI. Installed into a venv of its own inside
# the VM so Hermes' dependencies are never touched.
#
# https://github.com/taylorwilsdon/google_workspace_mcp (MIT)
#
# PINNED, and the pin is load-bearing. This add-on's guarantees are claims
# about one version's tool surface -- that no Drive delete or trash tool
# exists, and that Gmail tops out at gmail.modify. A floating version could
# add either in a release nobody here reviewed. verify.sh asserts the
# version AND re-derives the surface from the running server, so a bump
# fails loudly rather than silently widening what the agent can do.
#
# To bump: raise the version, run `addon-google-workspace.sh install`, then
# `verify --live`. If the tool-surface checks fail, read the upstream
# changelog before touching the assertions.
workspace-mcp==2.3.0

# Not optional here, despite being an optional dependency of httplib2.
#
# The Google API client talks through httplib2, whose connect() proxies
# only `if self.proxy_info and self.proxy_info.isgood() ...`, and
# isgood() is `return socks and ...`. Without PySocks that module global
# is None, so isgood() is falsy, httplib2 sets use_proxy = False and
# connects DIRECT -- straight into the uid lockdown, which answers
# [Errno 113] No route to host in about a quarter of a second.
#
# Everything looks right while this is missing: the proxy variables are
# set, squid is healthy, curl through it works, and the failing requests
# never appear in squid's log because they never reach it. That absence
# is the tell. Two agent sessions and ~100 retries were spent here.
PySocks==1.7.1
